-- CONSULTA VERIFICACION --

select * from con_comprobante as con
inner join ven_facdetalle as ven on (ven.tipfac=con.codtran)
                          and (ven.codcto=con.codcto)
						  and (ven.numfac= right(con.consec,8))
						  where ven.ivaart='19.00' and codtran='FV' and codcue='42950501' AND CREDITO > 100
