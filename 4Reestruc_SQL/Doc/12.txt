CREATE PROCEDURE dbo.db_SP_SYN_DATOSTRANSACCION_Inicializar
AS
BEGIN
	IF (OBJECT_ID('SYN_DATOSTRANSACCION','U') IS NULL)
	BEGIN
		CREATE TABLE SYN_DATOSTRANSACCION(
			am_LOG_TRAN NUMERIC(20) IDENTITY(1,1),
			cd_TIP_TRAN VARCHAR(200) NOT NULL DEFAULT 'I',
			cd_BD_ORIG  VARCHAR(300) NOT NULL DEFAULT '',
			cd_LOG_STAT VARCHAR(200) NOT NULL DEFAULT 'P',
			cd_TBL_NAME VARCHAR(300) NOT NULL DEFAULT '',
			xm_DAT_TRAN XML,
			xm_KEY_TRAN XML,
			dt_LOG_FECHA SMALLDATETIME DEFAULT GETDATE() NOT NULL,
			VersionA NUMERIC(20) NOT NULL DEFAULT 0)
		ALTER TABLE dbo.SYN_DATOSTRANSACCION ADD CONSTRAINT PK_SYN_DATOSTRANSACCION PRIMARY KEY(am_LOG_TRAN)
		ALTER TABLE dbo.SYN_DATOSTRANSACCION ADD CONSTRAINT CK_SYN_DATOSTRANSACCION_TIP_TRAN CHECK(cd_TIP_TRAN IN ('I','U','D'))
		ALTER TABLE dbo.SYN_DATOSTRANSACCION ADD CONSTRAINT CK_SYN_DATOSTRANSACCION_LOG_STAT CHECK(cd_LOG_STAT IN ('P','E','A'))
	END 
	IF (OBJECT_ID('SYN_REG_VERSIONESTABLA','U') IS NULL)
	BEGIN
		CREATE TABLE SYN_REG_VERSIONESTABLA(
			cd_SRV_NOMB VARCHAR(200) NOT NULL DEFAULT @@SERVERNAME,
			cd_TBL_NAME VARCHAR(200) NOT NULL DEFAULT '',
			am_NUM_VERS NUMERIC(10) NOT NULL DEFAULT 0,
			am_NUM_VERSSERVER NUMERIC(10)  NULL DEFAULT 0,
			dt_LOG_FECHA SMALLDATETIME DEFAULT GETDATE() NOT NULL)
		ALTER TABLE dbo.SYN_REG_VERSIONESTABLA ADD CONSTRAINT PK_SYN_REG_VERSIONESTABLA PRIMARY KEY(cd_SRV_NOMB,cd_TBL_NAME)
	END

			cd_SRV_NOMB VARCHAR(200) NOT NULL DEFAULT @@SERVERNAME,
			cd_TBL_NAME VARCHAR(200) NOT NULL DEFAULT '',
			am_NUM_VERS NUMERIC(10) NOT NULL DEFAULT 0,

CREATE PROCEDURE dbo.db_SP_SYN_DATOSTRANSACCION_inicializarTable
	@cd_TBL_NAME VARCHAR(200)
AS

	BEGIN
		SELECT 'Tabla ya se encuentra habilitada para Seguimiento de cambios'
	END
	ELSE
	BEGIN
		IF EXISTS(SELECT * FROM SYS.change_tracking_tables WHERE object_id=OBJECT_ID(@cd_TBL_NAME,'U'))
		BEGIN
			SELECT 'WARNING : El seguimiento de cambios ya se encuentra habilitado para esta base de datos'
		END
		ELSE
		BEGIN
			DECLARE @ds_CadenaSQL NVARCHAR(4000)
			SET @ds_CadenaSQL = 'ALTER TABLE dbo.' + @cd_TBL_NAME + ' ENABLE CHANGE_TRACKING'
			EXEC (@ds_CadenaSQL)
		END
		INSERT INTO dbo.SYN_REG_VERSIONESTABLA(cd_SRV_NOMB,cd_TBL_NAME,am_NUM_VERS,dt_LOG_FECHA)
			VALUES(@@SERVERNAME,@cd_TBL_NAME,0,GETDATE())
	END

CREATE PROCEDURE dbo.db_SP_SYN_DATOSTRANSACCION_Cambiosprocesados
@SERVERNAME VARCHAR(3000)
AS
BEGIN
	SET NOCOUNT ON;
	DECLARE curTablas CURSOR
		FOR SELECT cd_TBL_NAME FROM dbo.SYN_REG_VERSIONESTABLA
	DECLARE @cd_TBL_NAME VARCHAR(30)
	OPEN curTablas 
	FETCH NEXT FROM curTablas InTO @cd_TBL_NAME
	DECLARE @ds_Mensaje VARCHAR(200)
	WHILE @@FETCH_STATUS=0
	BEGIN
		SET @ds_Mensaje = 'Procesando tabla : ' + @cd_TBL_NAME
		RAISERROR(@ds_Mensaje,10,1) WITH NOWAIT
		EXEC db_SP_SYN_DATOSTRANSACCION_Enviodecambios @cd_TBL_NAME,@SERVERNAME
		FETCH NEXT FROM curTablas InTO @cd_TBL_NAME
	END
	CLOSE curTablas
	DEALLOCATE curTablas
END
GO

CREATE PROCEDURE dbo.db_SP_SYN_DATOSTRANSACCION_Enviodecambios
	@cd_TBL_NAME VARCHAR(20),
	@SERVERNAME VARCHAR(3000)
AS
BEGIN
	IF NOT EXISTS(SELECT * FROM dbo.SYN_REG_VERSIONESTABLA WHERE cd_TBL_NAME = @cd_TBL_NAME)
	BEGIN 
		SELECT 'No se encontro configuracion para Seguimiento de Cambios'
	END
	ELSE
	BEGIN
		IF (dbo.db_FC_SYN_DATOSTRANSACCION_Enviarllave(@cd_TBL_NAME,'llave') IS NULL)
		BEGIN
			SELECT 'Error al cargar informacion de llave primaria'
			RETURN
		END

		DECLARE @am_VersionDB NUMERIC(10)
		DECLARE @am_VersionAct NUMERIC(10)		
		SELECT @am_VersionDB = CHANGE_TRACKING_CURRENT_VERSION()
		SELECT  @am_VersionAct= am_NUM_VERS FROM SYN_REG_VERSIONESTABLA WHERE cd_SRV_NOMB=@@servername AND cd_TBL_NAME=@cd_TBL_NAME
		
		DECLARE @ds_CadenaSQL VARCHAR(8000)
								SET @ds_CadenaSQL = '
				SELECT CHANG.SYS_CHANGE_OPERATION ,''' + @cd_TBL_NAME + ''' , @@SERVERNAME, (SELECT '+dbo.db_FC_SYN_DATOSTRANSACCION_Enviarllave(@cd_TBL_NAME,'llave')+'
						FROM dbo.'+@cd_TBL_NAME+' llave
						WHERE '+dbo.db_FC_SYN_DATOSTRANSACCION_EnviarllaveJOIN(@cd_TBL_NAME,'llave','TABLA')+'
						FOR XML AUTO,TYPE, ELEMENTS) AS llave
						,(SELECT *,'''+@SERVERNAME+''' AS cd_BD_ORIG, 1500 AS am_LOG_TRAN
						FROM dbo.'+@cd_TBL_NAME+' reg
						WHERE '+dbo.db_FC_SYN_DATOSTRANSACCION_EnviarllaveJOIN(@cd_TBL_NAME,'reg','TABLA')+'
						FOR XML AUTO,TYPE, ELEMENTS) AS xmlData,
						CHANG.SYS_CHANGE_VERSION
				FROM dbo.' + @cd_TBL_NAME + ' TABLA
				INNER JOIN CHANGETABLE(CHANGES '+@cd_TBL_NAME+','+ CONVERT(VARCHAR(10),@am_VersionAct) + ') AS CHANG ON (' + dbo.db_FC_SYN_DATOSTRANSACCION_EnviarllaveJOIN(@cd_TBL_NAME,'TABLA','CHANG') + ')
				WHERE CHANG.SYS_CHANGE_VERSION BETWEEN ' + CONVERT(VARCHAR(10),@am_VersionAct) + ' AND '+ CONVERT(VARCHAR(10),@am_VersionDB)
				+ 'ORDER BY CHANG.SYS_CHANGE_VERSION'
		BEGIN TRY
			BEGIN TRANSACTION
				INSERT INTO SYN_DATOSTRANSACCION(cd_TIP_TRAN,cd_TBL_NAME,cd_BD_ORIG,xm_KEY_TRAN,xm_DAT_TRAN,VersionA)
				EXEC (@ds_CadenaSQL)
				RAISERROR(@ds_CadenaSQL,10,1) WITH NOWAIT
				UPDATE SYN_REG_VERSIONESTABLA SET am_NUM_VERS = @am_VersionDB WHERE						cd_TBL_NAME=@cd_TBL_NAME
				print @ds_CadenaSQL
				DECLARE @CANTIDAD INT
				DECLARE @CANT INT
				SET @CANTIDAD = (SELECT MAX(AM_LOG_TRAN) FROM SYN_DATOSTRANSACCION)
				SET @CANT = 1
				WHILE @CANT <= @CANTIDAD
				BEGIN
					DECLARE @ID INT
					DECLARE @SER VARCHAR(300)
					SET @ID = (SELECT AM_LOG_TRAN FROM SYN_DATOSTRANSACCION WHERE am_LOG_TRAN = @CANT)
						update SYN_DATOSTRANSACCION set xm_DAT_TRAN.modify('replace value of (/reg/am_LOG_TRAN/text())[1] with							sql:variable		("@ID")') , cd_BD_ORIG = 
						xm_DAT_TRAN.value('(/reg/servidor/text())[1]', 'VARCHAR(3000)' )   WHERE am_LOG_TRAN = @CANT 
					SET @SER = (SELECT xm_DAT_TRAN.value('(/reg/servidor/text())[1]', 'VARCHAR(3000)' ) FROM							SYN_DATOSTRANSACCION WHERE am_LOG_TRAN = @ID AND cd_TIP_TRAN = 'U')
					IF @SER = 'SERVER'
					BEGIN
						update SYN_DATOSTRANSACCION set xm_DAT_TRAN.modify('replace value of (/reg/am_LOG_TRAN/text())[1] with							sql:variable		("@ID")') , cd_BD_ORIG = 
						xm_DAT_TRAN.value('(/reg/cd_BD_ORIG/text())[1]', 'VARCHAR(3000)' )   WHERE am_LOG_TRAN = @CANT 
					END
					ELSE
					BEGIN
						update SYN_DATOSTRANSACCION set xm_DAT_TRAN.modify('replace value of (/reg/am_LOG_TRAN/text())[1] with					sql:variable		("@ID")') , cd_BD_ORIG = 
xm_DAT_TRAN.value('(/reg/servidor/text())[1]', 'VARCHAR(3000)' )   WHERE am_LOG_TRAN = @CANT 
					END
					PRINT @CANT
					SET @CANT = @CANT +1
				END
			COMMIT
		END TRY
		BEGIN CATCH
			SELECT 
				ERROR_NUMBER() AS ErrorNumber
				,ERROR_MESSAGE() AS ErrorMessage;
			ROLLBACK
		END CATCH
	END
END

CREATE FUNCTION dbo.db_FC_SYN_DATOSTRANSACCION_Enviarcolumnas(
	@cd_TBL_NAME VARCHAR(20) )
RETURNS VARCHAR(4000)
AS
BEGIN
	DECLARE @cd_COL_DEFI VARCHAR(8000)
	SELECT @cd_COL_DEFI = COALESCE(@cd_COL_DEFI + ' , [' + COLUMN_NAME  + ']' ,COLUMN_NAME)
		FROM INFORMATION_SCHEMA.COLUMNS
		WHERE TABLE_NAME = @cd_TBL_NAME
		AND TABLE_CATALOG = DB_NAME()
		ORDER BY ORDINAL_POSITION
	RETURN (IsNull(@cd_COL_DEFI,''))
END

CREATE FUNCTION dbo.db_FC_SYN_DATOSTRANSACCION_Enviarcolumnadefinicion(
	@cd_TBL_NAME VARCHAR(20) ,
	@cd_COL_NAME VARCHAR(200))
RETURNS VARCHAR(4000)
AS
BEGIN
	DECLARE @ds_Definicion VARCHAR(2000)
	DECLARE @ds_DataType VARCHAR(40)
	DECLARE @am_CharacterMaximumLength NUMERIC(10)
	DECLARE @am_NumericPrecision NUMERIC(10)
	DECLARE @am_NumericScale NUMERIC(10)

	SELECT @ds_DataType = DATA_TYPE
		, @am_CharacterMaximumLength = CHARACTER_MAXIMUM_LENGTH 
		, @am_NumericPrecision = NUMERIC_PRECISION
		, @am_NumericScale = NUMERIC_SCALE
		FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_NAME = @cd_TBL_NAME
		AND TABLE_CATALOG = DB_NAME()
		AND COLUMN_NAME = @cd_COL_NAME
	IF (@ds_DataType IS NULL)
	BEGIN
		SET @ds_Definicion=''
	END
	ELSE
	BEGIN
		SET @ds_Definicion='[' + @cd_COL_NAME+'] ' + @ds_DataType
		IF (NOT @ds_DataType IN('SMALLDATETIME','DATETIME','DATE','BIT'))
		BEGIN
			IF (@ds_DataType IN('NUMERIC','DECIMAL'))
			BEGIN
				SET @ds_Definicion = @ds_Definicion + '(' + CONVERT(VARCHAR(10),@am_NumericPrecision) + ',' + CONVERT(VARCHAR(10),@am_NumericScale) + ')'
			END
			ELSE
			BEGIN
				IF (@ds_DataType IN('UNIQUEIDENTIFIER','FLOAT'))
				BEGIN
					SET @ds_Definicion = @ds_Definicion 
				END
				ELSE
				BEGIN
					IF (@ds_DataType IN('NVARCHAR','VARCHAR') AND @am_CharacterMaximumLength=-1)
					BEGIN
						SET @ds_Definicion = @ds_Definicion + '(MAX)'
					END
					ELSE
					BEGIN
						SET @ds_Definicion = @ds_Definicion + '(' + CONVERT(VARCHAR(10),@am_CharacterMaximumLength) + ')'
					END
				END
			END
		END
	END
	RETURN (IsNull(@ds_Definicion,''))
END
GO

CREATE FUNCTION dbo.db_FC_SYN_DATOSTRANSACCION_Enviarllave(
	@cd_TBL_NAME VARCHAR(20),
	@ds_ALI_NAME VARCHAR(20) )
RETURNS VARCHAR(4000)
AS
BEGIN
	DECLARE @cd_KEY VARCHAR(1000)
	SELECT @cd_KEY = COALESCE(@cd_KEY+', '+'[Table].'+COLUMN_NAME,'[Table].'+COLUMN_NAME)
		FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE
		WHERE CONSTRAINT_NAME LIKE 'PK_' + @cd_TBL_NAME
		AND TABLE_CATALOG = DB_NAME()
		ORDER BY ORDINAL_POSITION
	SET @cd_KEY = REPLACE(@cd_KEY,'[Table]','['+@ds_ALI_NAME+']')
	RETURN (IsNull(@cd_KEY,''))
END
GO

CREATE FUNCTION dbo.db_FC_SYN_DATOSTRANSACCION_EnviarllaveJOIN(
	@cd_TBL_NAME VARCHAR(20),
	@ds_ALI_NAME1 VARCHAR(20),
	@ds_ALI_NAME2 VARCHAR(20))
RETURNS VARCHAR(4000)
AS
BEGIN
	DECLARE @cd_KEY VARCHAR(1000)
	SELECT @cd_KEY = COALESCE(@cd_KEY+' AND [Table1].'+COLUMN_NAME+'= [Table2].'+COLUMN_NAME,'[Table1].'+COLUMN_NAME+'= [Table2].'+COLUMN_NAME)
		FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE
		WHERE CONSTRAINT_NAME LIKE 'PK_' + @cd_TBL_NAME	
		AND TABLE_CATALOG = DB_NAME()
		ORDER BY ORDINAL_POSITION
	SET @cd_KEY = REPLACE(@cd_KEY,'[Table1]','['+@ds_ALI_NAME1+']')
	SET @cd_KEY = REPLACE(@cd_KEY,'[Table2]','['+@ds_ALI_NAME2+']')
	RETURN (IsNull(@cd_KEY,''))
END
